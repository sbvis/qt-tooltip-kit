cmake_minimum_required(VERSION 3.18)
project(QtTooltipKit VERSION 0.1.0 LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(QtTooltipKit INTERFACE)
add_library(QtTooltipKit::Tooltips ALIAS QtTooltipKit)
target_compile_features(QtTooltipKit INTERFACE cxx_std_17)
target_include_directories(QtTooltipKit INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(QtTooltipKit INTERFACE Qt6::Widgets)

option(QT_TOOLTIP_KIT_BUILD_TESTS "Build tooltip tests" OFF)
if(QT_TOOLTIP_KIT_BUILD_TESTS)
    enable_testing()
    add_executable(QtTooltipKitMarkupTest tests/MarkupTest.cpp)
    target_link_libraries(QtTooltipKitMarkupTest PRIVATE QtTooltipKit::Tooltips)
    add_test(NAME QtTooltipKitMarkupTest COMMAND QtTooltipKitMarkupTest)
endif()
